             /// <summary>
    /// Handles the SelectionChanged event to provide a context menu on non-mobile platforms.
    * </summary>
    /// <param name="sender">The SfDataGrid.</param>
    /// <param name="e">Event arguments containing the selected item.</param>
    private async void DataGrid_SelectionChanged(object sender, DataGridSelectionChangedEventArgs e)
    {
        // This approach is ideal for desktop platforms where right-click context menus are common.
        // On mobile, swipe gestures are more intuitive, so we can conditionally skip this.
#if WINDOWS || MACCATALYST
        if (e.AddedRows.Any() && this.BindingContext is GenericDataGridViewModel viewModel)
        {
            var selectedItem = e.AddedRows[0]; // Get the selected item

            // Display an action sheet to the user with Edit and Delete options
            string action = await DisplayActionSheet(
                "Choose Action", 
                "Cancel", 
                null, 
                "Edit", "Delete");

            switch (action)
            {
                case "Edit":
                    // Execute the Edit command from the ViewModel
                    if (viewModel.EditCommand.CanExecute(selectedItem))
                    {
                        viewModel.EditCommand.Execute(selectedItem);
                    }
                    break;

                case "Delete":
                    // Execute the Delete command from the ViewModel
                    if (viewModel.DeleteCommand.CanExecute(selectedItem))
                    {
                        viewModel.DeleteCommand.Execute(selectedItem);
                    }
                    break;
            }

            // De-select the row after the action to allow re-selection of the same row
            dataGrid.SelectedItem = null;
        }
#endif
    }

            // Use Shell navigation to go to the DeletePage.

    /// <summary>
    /// Handles the SelectionChanged event to provide a context menu on non-mobile platforms.
    /// This code will only be compiled for Windows and MacCatalyst.
    /// </summary>
    /// <param name="sender">The SfDataGrid.</param>
    /// <param name="e">Event arguments containing the selected item.</param>
    private async void DataGrid_SelectionChanged(object sender, DataGridSelectionChangedEventArgs e)
    {
        // The #if directive ensures this code only runs on desktop platforms
        // where swipe gestures are not the primary input method.
#if WINDOWS || MACCATALYST
        // Ensure that a row was actually added to the selection and the ViewModel exists.
        if (e.AddedRows.Any() && this.BindingContext is GenericDataGridViewModel viewModel)
        {
            // Get the data object from the selected row.
            var selectedItem = e.AddedRows[0]; 

            // Display a native action sheet to the user, which acts as a context menu.
            string action = await DisplayActionSheet(
                "Choose Action", // Title
                "Cancel",        // Cancel button text
                null,            // Destruction button text (e.g., for a destructive action)
                "Edit", "Delete"); // Option buttons

            // Based on the user's choice, execute the appropriate command from the ViewModel.
            switch (action)
            {
                case "Edit":
                    // Check if the command can be executed before invoking it.
                    if (viewModel.EditCommand.CanExecute(selectedItem))
                    {
                        viewModel.EditCommand.Execute(selectedItem);
                    }
                    break;

                case "Delete":
                    // Check if the command can be executed before invoking it.
                    if (viewModel.DeleteCommand.CanExecute(selectedItem))
                    {
                        viewModel.DeleteCommand.Execute(selectedItem);
                    }
                    break;
            }

            // Important: Clear the selection after the action is handled.
            // This allows the user to click and select the same row again to trigger the menu.
            if (sender is SfDataGrid grid)
            {
                grid.SelectedItem = null;
            }
        }
#endif
    }
